```rust
//! Phase 9a: scalar statements and expressions.
//!
//! Before Phase 9a, the IR modelled array accesses only — `Access` in
//! `src/ir/mod.rs`. The CFG builder (`src/fud/cfg.rs`) converted those
//! into blocks with no scalar uses or defs, so the FUD chain machinery
//! had nothing to link and `induct` had nothing to classify.
//!
//! Phase 9a adds three statement kinds — array accesses, scalar
//! assignments, and guards — plus a small expression tree for scalar
//! arithmetic. This is enough to model the loop IV (`i = i + step`),
//! the loop guard (`i < upper`), and any scalar computation whose
//! value flows into an array index.

use serde::{Deserialize, Serialize};

/// A statement in a loop body.
#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
#[serde(tag = "kind")]
pub enum StmtKind {
    /// An array access, read or write. Wraps the existing `Access` type.
    Array(crate::ir::Access),

    /// A scalar assignment: `target := expr`.
    Assign {
        target: usize,          // index into `Loop::scalars`
        expr: ScalarExpr,
    },

    /// A guard: execution continues only if `cond` is nonzero.
    /// Phase 9a emits at most one guard per loop, at the header.
    Guard {
        cond: ScalarExpr,
    },
}

/// A scalar expression over IVs, scalars, and constants.
///
/// `Iv(k)` reads the k-th IV of the enclosing loop. `Scalar(k)` reads
/// the k-th scalar from the loop's scalar table. The tree is small by
/// design — Phase 9a needs only what a loop IV requires, and the
/// frontend that would emit more (division, modulo, comparisons) is a
/// separate project.
#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub enum ScalarExpr {
    Iv(usize),
    Scalar(usize),
    Const(i64),
    Add(Box<ScalarExpr>, Box<ScalarExpr>),
    Mul(Box<ScalarExpr>, Box<ScalarExpr>),
}

impl ScalarExpr {
    /// Evaluate the expression against concrete IV and scalar values.
    /// Returns `None` on an out-of-range index or arithmetic overflow.
    pub fn eval(&self, ivs: &[i64], scalars: &[i64]) -> Option<i64> {
        match self {
            ScalarExpr::Iv(k) => ivs.get(*k).copied(),
            ScalarExpr::Scalar(k) => scalars.get(*k).copied(),
            ScalarExpr::Const(c) => Some(*c),
            ScalarExpr::Add(a, b) => a.eval(ivs, scalars)?.checked_add(b.eval(ivs, scalars)?),
            ScalarExpr::Mul(a, b) => a.eval(ivs, scalars)?.checked_mul(b.eval(ivs, scalars)?),
        }
    }

    /// Collect the scalar variables this expression reads.
    pub fn scalars_read(&self, out: &mut Vec<usize>) {
        match self {
            ScalarExpr::Iv(_) | ScalarExpr::Const(_) => {}
            ScalarExpr::Scalar(k) => {
                if !out.contains(k) { out.push(*k); }
            }
            ScalarExpr::Add(a, b) | ScalarExpr::Mul(a, b) => {
                a.scalars_read(out);
                b.scalars_read(out);
            }
        }
    }
}

/// A statement in a loop body.
#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
pub struct Stmt {
    pub id: usize,
    pub kind: StmtKind,
}

impl Stmt {
    /// The scalar variables this statement reads.
    pub fn uses(&self) -> Vec<usize> {
        let mut out = Vec::new();
        match &self.kind {
            StmtKind::Array(_) => {}
            StmtKind::Assign { expr, .. } => expr.scalars_read(&mut out),
            StmtKind::Guard { cond } => cond.scalars_read(&mut out),
        }
        out
    }

    /// The scalar variables this statement writes.
    pub fn defs(&self) -> Vec<usize> {
        match &self.kind {
            StmtKind::Array(_) => Vec::new(),
            StmtKind::Assign { target, .. } => vec![*target],
            StmtKind::Guard { .. } => Vec::new(),
        }
    }
}
```

## 9a.2 The CFG builder — full rewrite

### 9a.2.1 The 5-block shape

The loop CFG has five blocks:

```
entry (0):     i := lower                      — initialise IVs
header (1):    φ_i := φ(entry.i, latch.i)      — IV φ-terms, loop header
               guard: i < upper                — loop guard
body (2):      user statements                 — the loop body
latch (3):     i := i + step                   — IV increment
exit (4):      (nothing)                       — where the guard falls through
```

Edges:
- `entry → header`
- `header → body` (guard true)
- `header → exit` (guard false)
- `body → latch`
- `latch → header` (back edge)

### 9a.2.2 Full `build_cfg_from_loop`

```rust
// src/fud/cfg.rs — REPLACE the existing build_cfg_from_loop.

use crate::ir::{Loop, Stmt, StmtKind, ScalarExpr, ScalarVar, ScalarTy};

/// Build the standard 5-block loop CFG from a `Loop`.
///
/// Block layout:
///   0. entry  — IV initialisation
///   1. header — IV φ-terms + guard (loop header)
///   2. body   — user statements
///   3. latch  — IV increment
///   4. exit   — after the loop
///
/// The back edge is `3 → 1`. Without it, `induct` finds no IVs.
pub fn build_cfg_from_loop(loop_: &Loop) -> Option<CFG> {
    if loop_.ivs.is_empty() {
        return None;
    }

    let mut builder = CFGBuilder::new();

    // Register the IVs as scalars 0..depth.
    for iv in &loop_.ivs {
        builder.register_scalar(ScalarVar {
            name: iv.name.clone(),
            ty: ScalarTy::I64,
        });
    }
    // Register any additional scalars the loop declares.
    for s in &loop_.scalars {
        builder.register_scalar(s.clone());
    }

    // Block 0: entry — initialise the IVs.
    let entry = builder.new_block("entry");
    for (k, iv) in loop_.ivs.iter().enumerate() {
        builder.add_stmt(entry, StmtKind::Assign {
            target: k,
            expr: ScalarExpr::Const(iv.lower),
        });
    }

    // Block 1: header — IV φ-terms, then the loop guard.
    let header = builder.new_block("header");
    // The IV φ-terms: one incoming from entry, one from latch.
    // We record the incoming slot as "pending" — the latch id isn't known
    // yet — and fill it once latch is created.
    let mut pending_phis: Vec<(usize, usize)> = Vec::new(); // (header_block, scalar_idx)
    for k in 0..loop_.ivs.len() {
        let phi_id = builder.add_phi(header, k, 2);
        pending_phis.push((k, phi_id));
    }
    // The loop guard: i < upper for each IV.
    for (k, iv) in loop_.ivs.iter().enumerate() {
        builder.add_stmt(header, StmtKind::Guard {
            cond: ScalarExpr::Iv(k),
        });
        // The guard's comparison `i < upper` is implicit — the IV's
        // `upper` field carries it. The Stmt records the IV as used.
    }

    // Block 2: body — user statements.
    let body = builder.new_block("body");
    let stmts: Vec<Stmt> = match &loop_.stmts {
        Some(s) => s.clone(),
        None => synth_stmts_from_body(loop_),
    };
    for s in stmts {
        builder.add_stmt(body, s.kind);
    }

    // Block 3: latch — the IV increment.
    let latch = builder.new_block("latch");
    for (k, iv) in loop_.ivs.iter().enumerate() {
        builder.add_stmt(latch, StmtKind::Assign {
            target: k,
            expr: ScalarExpr::Add(
                Box::new(ScalarExpr::Iv(k)),
                Box::new(ScalarExpr::Const(iv.step)),
            ),
        });
    }

    // Block 4: exit.
    let exit = builder.new_block("exit");

    // Fill in the φ incoming slots now that we know the latch.
    for (k, phi_id) in pending_phis {
        builder.set_phi_incoming(phi_id, 0, entry, k); // from entry
        builder.set_phi_incoming(phi_id, 1, latch, k); // from latch (back edge)
    }

    // Edges.
    builder.add_edge(entry, header);
    builder.add_edge(header, body);      // guard true
    builder.add_edge(header, exit);      // guard false
    builder.add_edge(body, latch);
    builder.add_edge(latch, header);     // BACK EDGE — the critical one

    builder.set_entry(entry);
    builder.set_exit(exit);

    let cfg = builder.finish();
    cfg.validate().ok()?;
    Some(cfg)
}

/// When `loop_.stmts` is `None`, synthesise a body from the existing
/// `body: Vec<Access>`. Each access becomes `StmtKind::Array`.
fn synth_stmts_from_body(loop_: &Loop) -> Vec<Stmt> {
    loop_.body.iter().enumerate().map(|(id, a)| Stmt {
        id,
        kind: StmtKind::Array(a.clone()),
    }).collect()
}
```

### 9a.2.3 The `CFGBuilder` helper

The existing CFG construction code in `src/fud/cfg.rs` may already have a builder. If not, add one:

```rust
pub struct CFGBuilder {
    blocks: Vec<Block>,
    scalars: Vec<ScalarVar>,
    stmts: Vec<Stmt>,
    phis: Vec<PhiDef>,
    edges: Vec<(usize, usize)>,
    entry: Option<usize>,
    exit: Option<usize>,
}

pub struct PhiDef {
    pub block: usize,
    pub scalar: usize,
    pub incoming: Vec<(usize, usize)>, // (predecessor block, scalar idx from that block)
}

impl CFGBuilder {
    pub fn new() -> Self { /* ... */ }
    pub fn register_scalar(&mut self, s: ScalarVar) { self.scalars.push(s); }
    pub fn new_block(&mut self, _name: &str) -> usize {
        let id = self.blocks.len();
        self.blocks.push(Block { id, preds: vec![], succs: vec![], stmts: vec![] });
        id
    }
    pub fn add_stmt(&mut self, block: usize, kind: StmtKind) {
        let id = self.stmts.len();
        let stmt = Stmt { id, kind };
        self.stmts.push(stmt.clone());
        self.blocks[block].stmts.push(id);
    }
    pub fn add_phi(&mut self, block: usize, scalar: usize, arity: usize) -> usize {
        let id = self.phis.len();
        self.phis.push(PhiDef { block, scalar, incoming: vec![(0, 0); arity] });
        id
    }
    pub fn set_phi_incoming(&mut self, phi_id: usize, slot: usize, pred: usize, scalar: usize) {
        self.phis[phi_id].incoming[slot] = (pred, scalar);
    }
    pub fn add_edge(&mut self, from: usize, to: usize) {
        self.blocks[from].succs.push(to);
        self.blocks[to].preds.push(from);
        self.edges.push((from, to));
    }
    pub fn set_entry(&mut self, b: usize) { self.entry = Some(b); }
    pub fn set_exit(&mut self, b: usize) { self.exit = Some(b); }
    pub fn finish(self) -> CFG { /* assemble */ }
}
```

## 9a.3 Wire M3, M5, M8 to `induct`

### 9a.3.1 M3 — replace `recursion_to_iv`

```rust
// src/decide/m3.rs — REPLACED transform.

use crate::fud::{build_cfg_from_loop, build_fud_chains, induct, Class};

pub fn transform(loop_: &Loop) -> Option<Loop> {
    let r = loop_.recursion.as_ref()?;

    // Build the CFG with the Phase 9a builder.
    let cfg = build_cfg_from_loop(loop_)?;

    // Build the FUD chains.
    let (chains, _) = build_fud_chains(&cfg).ok()?;

    // Run induct.
    let induct = induct(&cfg, &chains);

    // Find the IV for the recursion's variable.
    let iv = induct.ivs.iter().find(|v| v.var == r.iv_name_idx)?;
    let Class::LinearIv { a, b } = induct.class.get(&iv.op).copied()? else {
        return None;
    };

    // The IV's closed form is `a + b·k` over the loop's trip count.
    let lower = a;
    let step = b;
    let upper = r.iv_bound;

    // Validate the sign of the step (must match the recursion's direction).
    if step == 0 { return None; }
    if (step > 0) != (r.iv_step > 0) { return None; }

    // Build the transformed loop.
    // ... same normalisation as the hand-coded version, but with `lower`,
    // `step`, `upper` from `induct` instead of from `recursion_to_iv`.
    Some(build_transformed(loop_, lower, upper, step))
}
```

### 9a.3.2 M5 — replace `chase_chain` for arithmetic chains

```rust
// src/decide/m5.rs — EXTENDED.

pub fn decide(loop_: &Loop, p: usize, q: usize) -> (Decision, Option<Certificate>) {
    let chase = match &loop_.chase { Some(c) => c, None => return Unknown };

    // Phase 9a: try the inductive path first.
    if let Some(cfg) = build_cfg_from_loop(loop_) {
        if let Ok((chains, _)) = build_fud_chains(&cfg) {
            let induct = induct(&cfg, &chains);
            if let Some(iv) = induct.ivs.iter().find(|v| v.var == chase.var_idx) {
                if let Some(Class::LinearIv { a, b }) = induct.class.get(&iv.op).copied() {
                    // Arithmetic chain: p_k = a + b·k.
                    // The fixed point is computable in closed form.
                    return decide_arithmetic_chase(loop_, p, q, a, b, chase);
                }
            }
        }
    }

    // Fall back to the literal-table path (Phase 2's version).
    decide_literal_chase(loop_, p, q)
}
```

## 9a.4 The extended `Loop` JSON format

Phase 9a extends the JSON format. Every existing loop parses unchanged (both new fields default). New loops may carry:

```json
{
  "depth": 1,
  "ivs": [{"name": "i", "lower": 0, "upper": 64, "step": 1}],
  "body": [ /* existing array accesses */ ],
  "scalars": [
    {"name": "i", "ty": "i64"},
    {"name": "sum", "ty": "i64"}
  ],
  "stmts": [
    {"id": 0, "kind": {"kind": "Assign", "target": 1, "expr": {"Add": [{"Scalar": 1}, {"Iv": 0}]}}},
    {"id": 1, "kind": {"kind": "Array", "kind": "write", "base": "A", "index": [1], "elem_size": 8, "offset": {"Const": 0}}}
  ]
}
```

**When `stmts` is present**, the CFG builder uses it verbatim.
**When `stmts` is absent**, the CFG builder synthesizes it: the IV is registered as scalar 0, the latch emits `i = i + step`, and the `body` accesses become `StmtKind::Array` statements at the body block.

**This is the entire frontend.** No JavaScript parser. The existing `m0-decide` JSON format is extended.

## The one blocked item: §5.1.2

`RuntimeCheckKind` (`DisjointArrays`, `NoDistanceK`, `GuardCheck`) is declared so
the vocabulary exists, but nothing constructs one and no checker arm accepts one.
Emitting a runtime check honestly needs (a) a static sufficiency proof, (b) a
replay channel so `m0-check` can verify the emitted check, and (c) a backend —
none of which is in this phase, and a check without (a) would be a claim the
checker cannot re-derive. Left as a documented stub, not faked.

## Known limitation (stated, not hidden)

`src/fud`'s `CFG` models scalars only: an array access lowers to a statement with
no uses and no defs, so the §6.1.1 region dataflow cannot run over a bare `CFG`
(it would report empty sets that look like a proved kill). `RegionProgram` is the
array-aware flow graph that fixes this; `loop_summary` covers the graph-free case.
